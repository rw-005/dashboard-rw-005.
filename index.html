/** * SISTEM CLOUD DATABASE KAS & KEPENDUDUKAN RW 05 * Backend: Google Apps Script + Google Sheets */ // 1. FUNGSI INISIALISASI STRUKTUR TABEL function setupSheets() { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Tab Kas var sKas = ss.getSheetByName("Kas"); if (!sKas) { sKas = ss.insertSheet("Kas"); sKas.appendRow(["ID", "Tanggal", "Tipe", "Wilayah", "Kategori", "Keterangan", "Nominal"]); } // Tab Warga var sWarga = ss.getSheetByName("Warga"); if (!sWarga) { sWarga = ss.insertSheet("Warga"); sWarga.appendRow([ "ID", "RT", "NoKK", "GenderKK", "NamaKK", "NikKK", "NamaIstri", "NikIstri", "JmlAnakLaki", "NamaAnakLaki", "JmlAnakP", "NamaAnakP", "StatusEkonomi", "IsRutilahu", "IsBansos", "JenisBansos", "Foto" ]); } // Tab Dinamika (Ngontrak, Lahir, Wafat) var sDinamika = ss.getSheetByName("Dinamika"); if (!sDinamika) { sDinamika = ss.insertSheet("Dinamika"); sDinamika.appendRow(["ID", "Bulan", "RT", "KKKontrak", "LakiKontrak", "PerempuanKontrak", "Meninggal", "Lahir"]); } // Tab Kegiatan & Foto var sKegiatan = ss.getSheetByName("Kegiatan"); if (!sKegiatan) { sKegiatan = ss.insertSheet("Kegiatan"); sKegiatan.appendRow(["ID", "Tanggal", "Wilayah", "Nama", "Lokasi", "Deskripsi", "Foto"]); } } // 2. FUNGSI TARIK DATA (doGet) - DIJALANKAN KETIKA WEB DIBUKA function doGet(e) { setupSheets(); var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = { status: "success", transaksi: getSheetDataNormalized(ss.getSheetByName("Kas")), warga: getSheetDataNormalized(ss.getSheetByName("Warga")), dinamika: getSheetDataNormalized(ss.getSheetByName("Dinamika")), kegiatan: getSheetDataNormalized(ss.getSheetByName("Kegiatan")) }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 3. FUNGSI SIMPAN DATA (doPost) - MENERIMA INPUT DARI FORM WEB function doPost(e) { setupSheets(); var ss = SpreadsheetApp.getActiveSpreadsheet(); try { var contents = e.postData.contents; var req = JSON.parse(contents); var action = req.action; var data = req.data; // A. Simpan Transaksi Kas Masuk / Keluar if (action === "tambah_kas") { var sKas = ss.getSheetByName("Kas"); sKas.appendRow([ data.id || Date.now(), data.tanggal || Utilities.formatDate(new Date(), "GMT+7", "yyyy-MM-dd"), data.tipe || "masuk", data.entitas || data.wilayah || "RT 01", data.kategori || "-", data.keterangan || "-", Number(data.nominal) || 0 ]); return jsonResponse({ status: "success", message: "Kas berhasil disimpan" }); } // B. Simpan Seluruh Data Induk Warga (Termasuk Import Excel & Edit) if (action === "simpan_warga_all") { var sWarga = ss.getSheetByName("Warga"); sWarga.clearContents(); sWarga.appendRow([ "ID", "RT", "NoKK", "GenderKK", "NamaKK", "NikKK", "NamaIstri", "NikIstri", "JmlAnakLaki", "NamaAnakLaki", "JmlAnakP", "NamaAnakP", "StatusEkonomi", "IsRutilahu", "IsBansos", "JenisBansos", "Foto" ]); if (Array.isArray(data)) { var rows = []; data.forEach(function(w) { rows.push([ w.id || Date.now(), w.rt || "RT 01", "'" + String(w.noKK || "-"), // Diberi kutip agar 16 digit tidak terpotong ilmiah w.genderKK || "L", w.namaKK || "-", "'" + String(w.nikKK || "-"), w.namaIstri || "-", "'" + String(w.nikIstri || "-"), Number(w.jmlAnakLaki) || 0, w.namaAnakLaki || "-", Number(w.jmlAnakP) || 0, w.namaAnakP || "-", w.statusEkonomi || "Sejahtera", w.isRutilahu ? true : false, w.isBansos ? true : false, w.jenisBansos || "-", w.foto || "" ]); }); if (rows.length > 0) { sWarga.getRange(2, 1, rows.length, rows[0].length).setValues(rows); } } return jsonResponse({ status: "success", message: "Data warga berhasil disinkronkan" }); } // C. Simpan Dinamika Kependudukan (Wafat, Lahir, Ngontrak) if (action === "simpan_dinamika_all") { var sDin = ss.getSheetByName("Dinamika"); sDin.clearContents(); sDin.appendRow(["ID", "Bulan", "RT", "KKKontrak", "LakiKontrak", "PerempuanKontrak", "Meninggal", "Lahir"]); if (Array.isArray(data) && data.length > 0) { var dinRows = data.map(function(d) { return [ d.id || Date.now(), d.bulan || "", d.rt || "RT 01", Number(d.kkKontrak) || 0, Number(d.lakiKontrak) || 0, Number(d.perempuanKontrak) || 0, Number(d.meninggal) || 0, Number(d.lahir) || 0 ]; }); sDin.getRange(2, 1, dinRows.length, dinRows[0].length).setValues(dinRows); } return jsonResponse({ status: "success", message: "Dinamika tersimpan" }); } // D. Simpan Dokumentasi Kegiatan if (action === "simpan_kegiatan_all") { var sKeg = ss.getSheetByName("Kegiatan"); sKeg.clearContents(); sKeg.appendRow(["ID", "Tanggal", "Wilayah", "Nama", "Lokasi", "Deskripsi", "Foto"]); if (Array.isArray(data) && data.length > 0) { var kegRows = data.map(function(k) { return [ k.id || Date.now(), k.tanggal || "", k.wilayah || "RW 05", k.nama || "", k.lokasi || "-", k.deskripsi || "-", k.foto || "" ]; }); sKeg.getRange(2, 1, kegRows.length, kegRows[0].length).setValues(kegRows); } return jsonResponse({ status: "success", message: "Kegiatan tersimpan" }); } return jsonResponse({ status: "error", message: "Action tidak dikenali" }); } catch (err) { return jsonResponse({ status: "error", message: err.toString() }); } } // 4. HELPER: BACA DATA DENGAN CASE-INSENSITIVE (AGAR SEMUA HURUF TERBACA) function getSheetDataNormalized(sheet) { if (!sheet) return []; var vals = sheet.getDataRange().getValues(); if (vals.length <= 1) return []; var headers = vals[0].map(function(h) { return String(h).trim().toLowerCase().replace(/[^a-z0-9]/g, ''); }); var rows = []; for (var i = 1; i < vals.length; i++) { var r = vals[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var key = headers[j]; var cellVal = r[j]; // Hilangkan tanda kutip pengaman teks jika ada if (typeof cellVal === "string" && cellVal.indexOf("'") === 0) { cellVal = cellVal.substring(1); } obj[key] = cellVal; } rows.push(obj); } return rows; } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
